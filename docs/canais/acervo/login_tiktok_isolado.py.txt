"""TikTok isolado, perfil proprio, com reforco anti-deteccao (mesmas
medidas do oauth-browser/app.py: esconder navigator.webdriver, user agent
realista, --disable-blink-features=AutomationControlled). TikTok derrubou
o processo inteiro quando estava junto com Facebook/Instagram -- isolado
aqui pra nao arriscar o que ja funcionou."""

import json
import sys
import time

from playwright.sync_api import sync_playwright

PROFILE_DIR = r"C:\Users\LUCAS~1.RAN\AppData\Local\Temp\chrome_login_tiktok"
CHROME_EXE = r"C:\Program Files\Google\Chrome\Application\chrome.exe"
SAIDA = r"C:\Users\LUCAS~1.RAN\AppData\Local\Temp\cookies_tiktok.json"

with sync_playwright() as p:
    ctx = p.chromium.launch_persistent_context(
        PROFILE_DIR,
        headless=False,
        executable_path=CHROME_EXE,
        args=[
            "--no-first-run",
            "--no-default-browser-check",
            "--disable-session-crashed-bubble",
            "--disable-blink-features=AutomationControlled",
        ],
        user_agent=(
            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
            "(KHTML, like Gecko) Chrome/128.0.0.0 Safari/537.36"
        ),
    )
    ctx.add_init_script(
        "Object.defineProperty(navigator, 'webdriver', {get: () => undefined})"
    )
    page = ctx.pages[0] if ctx.pages else ctx.new_page()
    print("indo pro tiktok...", file=sys.stderr, flush=True)
    try:
        page.goto("https://www.tiktok.com/login/phone-or-email/email", wait_until="commit", timeout=30000)
    except Exception as e:  # noqa: BLE001
        print(f"falha no goto inicial: {e}", file=sys.stderr, flush=True)

    print(">>> FACA LOGIN NO TIKTOK. Detectando ate 15 minutos.", file=sys.stderr, flush=True)

    inicio = time.time()
    achou = False
    while (time.time() - inicio) < 900:
        try:
            cks = [c for c in ctx.cookies() if "tiktok.com" in c["domain"]]
        except Exception as e:  # noqa: BLE001
            print(f"contexto morreu: {e}", file=sys.stderr, flush=True)
            break
        nomes = {c["name"] for c in cks}
        if "sid_tt" in nomes or "sessionid" in nomes:
            with open(SAIDA, "w", encoding="utf-8") as f:
                json.dump(cks, f, ensure_ascii=False, indent=2)
            print(f"[OK] tiktok logado -- {len(cks)} cookies salvos em {SAIDA}", file=sys.stderr, flush=True)
            achou = True
            break
        time.sleep(10)

    if not achou:
        print("tiktok NAO logou a tempo (ou o contexto caiu de novo)", file=sys.stderr, flush=True)

    try:
        ctx.close()
    except Exception:  # noqa: BLE001
        pass
